    InterleaverKind max_interleaver = InterleaverKind::Long;
    bool soft_decisions = false;
};

    None = 0,
    BurstStarted = 1,
    BlockDecoded = 2,
    BurstCompleted = 3,

    BurstRefused = 6,
};

    int waveform = -1;

    double cfo_hz = 0;

    std::span<const float> soft;
    BurstEndReason reason = BurstEndReason::Eom;
    int blocks = 0;
    std::int64_t payload_bits = 0;
};
